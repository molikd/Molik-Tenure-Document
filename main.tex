\newcommand{\evidenceref}[1]{%
  \,\hyperlink{evidence-#1}{%
    [#1\ifcsname evidencelabel@#1\endcsname
       \ifcsname ifevidenceshown@#1\endcsname
         \csname ifevidenceshown@#1\endcsname
           \else: \csname evidencelabel@#1\endcsname
             \expandafter\global\csname evidenceshown@#1true\endcsname
         \fi
       \fi
     \fi]}}

% Reserve margins around each complete source page for its evidence label.
% Source PDFs remain byte-identical; every page is included, without cropping.
\newcounter{evidencesourcepage}
\newcommand{\evidencepdf}[4]{%
  \clearpage
  \setcounter{evidencesourcepage}{0}%
  \includepdf[pages=-,scale=0.88,
    addtotoc={1,subsection,2,{#1 --- #2},supplement-#1},
    pagecommand={%
      \thispagestyle{empty}%
      \stepcounter{evidencesourcepage}%
      \AddToShipoutPictureFG*{%
        \AtPageUpperLeft{\put(0,-28){\makebox[\paperwidth]{%
          \small\bfseries Evidence #1 --- #2}}}%
        \AtPageLowerLeft{\put(0,18){\makebox[\paperwidth]{%
          \small Source page \arabic{evidencesourcepage} of #4
          \quad\textbar\quad Portfolio page \thepage}}}%
      }%
    }]{#3}%
}
