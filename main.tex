% Per-section header. moderncv's \makecvtitle allocates its layout lengths
% with \newcommand, so it can only be called once per document; the
% portfolio calls a header in every section, hence this repeatable
% replacement (name + section title + rule, matching the cover styling).
\makeatletter
\newcommand{\portfoliotitle}{%
  {\raggedright
   {\color{color1}\huge\bfseries \@firstname\ \@familyname\par}
   \vspace{0.1in}
   {\color{color1}\Large \@title\par}
   \vspace{0.1in}
   {\color{color1}\rule{\textwidth}{1pt}\par}
   \vspace{0.25in}}%
}
