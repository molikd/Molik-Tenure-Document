% Reserve margins around intact source pages; originals remain byte-identical.
% An optional explicit page list selects excerpts and preserves source numbering.
\ExplSyntaxOn
\NewExpandableDocumentCommand{\evidenceoriginalpage}{mm}{\clist_item:nn{#1}{#2}}
\ExplSyntaxOff
\newcounter{evidencesourcepage}
\newcommand{\evidencepdf}[5][-]{%
  \clearpage
  \begingroup
  \def\evidenceselection{#1}\def\evidenceallpages{-}%
  \ifx\evidenceselection\evidenceallpages
    \def\evidencefirstpage{1}%
    \def\evidencecurrentpage{\arabic{evidencesourcepage}}%
  \else
    \edef\evidencefirstpage{\evidenceoriginalpage{#1}{1}}%
    \def\evidencecurrentpage{\evidenceoriginalpage{#1}{\value{evidencesourcepage}}}%
  \fi
  \setcounter{evidencesourcepage}{0}%
  % pdfpages addtotoc matches the original PDF page, not its excerpt position.
  % For an excerpt starting at source page 2, using 1 would omit the TOC anchor.
  \includepdf[pages={#1},scale=0.88,
    addtotoc={\evidencefirstpage,subsection,2,{#2 --- #3},supplement-#2},
    pagecommand={%
      \thispagestyle{empty}%
      \stepcounter{evidencesourcepage}%
      \AddToShipoutPictureFG*{%
        \AtPageUpperLeft{\put(0,-28){\makebox[\paperwidth]{%
          \small\bfseries Evidence #2 --- #3}}}%
        \AtPageLowerLeft{\put(0,18){\makebox[\paperwidth]{%
          \small Source page \evidencecurrentpage\ of #5
          \quad\textbar\quad Portfolio page \thepage}}}%
      }%
    }]{#4}%
  \endgroup
}
